macro Instrument
if "\1" == "_"
    dw      DummyTable
else
    dw      vol_\1
endc
if "\2" == "_"
    dw      DummyTable
else
    dw      arp_\2
endc
if "\3" == "_"
    dw      DummyTable
else
    dw      waveseq_\3
endc
if "\4" == "_"
    dw      vib_Dummy
else
    dw      vib_\4
endc
    endm

macro Drum
    db      SetInstrument,_\1,fix,\2
    endm
